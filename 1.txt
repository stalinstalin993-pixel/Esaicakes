3. app.js (Dynamic Catalog & WhatsApp Generator)Save this in the root folder. Change WHATSAPP_NUMBER on Line 2 to your client's 10-digit mobile number with country code (91 for India).   JavaScript// REPLACE THIS WITH THE BAKERY OWNER'S WHATSAPP NUMBER (Include country code, no '+' sign)
const WHATSAPP_NUMBER = "919876543210";

// Data Source URL:
// Option A (Static Git Push): Keep as 'cakes.json'
// Option B (Google Sheets): Replace with your published Google Sheet CSV/JSON web link if needed later
const DATA_URL = "cakes.json";

let allCakes = [];

const cakeGrid = document.getElementById("cake-grid");
const filterButtons = document.querySelectorAll(".filter-btn");
const footerWaLink = document.getElementById("footer-wa-link");

// Set footer quick-chat link
footerWaLink.href = `https://wa.me/${WHATSAPP_NUMBER}?text=${encodeURIComponent("Hi! I have a query about ordering a cake.")}`;

// 1. Fetch Cakes from cakes.json
async function loadCakes() {
  try {
    const response = await fetch(DATA_URL);
    allCakes = await response.json();
    renderCakes(allCakes);
  } catch (error) {
    cakeGrid.innerHTML = `<p class="col-span-full text-center text-red-600">Unable to load cake menu. Please make sure cakes.json is uploaded.</p>`;
    console.error("Error loading cakes.json:", error);
  }
}

// 2. Render Cake Cards into the Grid
function renderCakes(cakes) {
  if (cakes.length === 0) {
    cakeGrid.innerHTML = `<p class="col-span-full text-center text-gray-500 py-8">No cakes found in this category.</p>`;
    return;
  }

  cakeGrid.innerHTML = cakes
    .map((cake) => {
      const waMessage = `Hi! I want to order the *${cake.name}* [Model #${cake.id}] priced at ${cake.price}. Is it available?`;
      const waUrl = `https://wa.me/${WHATSAPP_NUMBER}?text=${encodeURIComponent(waMessage)}`;

      return `
      <div class="bg-white rounded-2xl overflow-hidden shadow-sm hover:shadow-md transition border border-pink-100 flex flex-col">
        <div class="relative h-56 overflow-hidden bg-pink-50">
          <img src="${cake.image}" alt="${cake.name}" loading="lazy" class="w-full h-full object-cover hover:scale-105 transition duration-300" />
          <span class="absolute top-3 left-3 bg-white/90 backdrop-blur-sm text-cocoa text-xs font-bold px-2.5 py-1 rounded-full shadow-sm">
            #${cake.id}
          </span>
          ${
            !cake.available
              ? `<span class="absolute top-3 right-3 bg-red-600 text-white text-xs font-bold px-3 py-1 rounded-full">Sold Out</span>`
              : `<span class="absolute top-3 right-3 bg-pink-100 text-rose text-xs font-semibold px-2.5 py-1 rounded-full">${cake.category}</span>`
          }
        </div>

        <div class="p-4 flex-1 flex flex-col justify-between">
          <div>
            <h3 class="font-bold text-base text-cocoa line-clamp-1">${cake.name}</h3>
            <p class="text-rose font-extrabold text-lg mt-1">${cake.price}</p>
          </div>

          <div class="mt-4">
            ${
              cake.available
                ? `<a href="${waUrl}" target="_blank" class="w-full bg-green-600 hover:bg-green-700 text-white text-sm font-semibold py-2.5 px-4 rounded-xl flex items-center justify-center gap-2 transition">
                    <span>Order on WhatsApp</span>
                   </a>`
                : `<button disabled class="w-full bg-gray-200 text-gray-500 text-sm font-semibold py-2.5 px-4 rounded-xl cursor-not-allowed">
                    Currently Sold Out
                   </button>`
            }
          </div>
        </div>
      </div>
    `;
    })
    .join("");
}

// 3. Category Tab Filtering
filterButtons.forEach((btn) => {
  btn.addEventListener("click", () => {
    // Update active button styling
    filterButtons.forEach((b) => {
      b.classList.remove("bg-rose", "text-white");
      b.classList.add("bg-white", "text-cocoa");
    });
    btn.classList.remove("bg-white", "text-cocoa");
    btn.classList.add("bg-rose", "text-white");

    const selectedCategory = btn.getAttribute("data-category");
    if (selectedCategory === "All") {
      renderCakes(allCakes);
    } else {
      const filtered = allCakes.filter((cake) => cake.category === selectedCategory);
      renderCakes(filtered);
    }
  });
});

// 4. Custom Cake Inquiry Form -> WhatsApp Handler

🌱 *Preference:* ${preference}
📅 *Date Needed:* ${date}
✍️ *Message on Cake:* "${message}"
📝 *Notes:* ${notes}`;

  const whatsappUrl = `https://wa.me/${WHATSAPP_NUMBER}?text=${encodeURIComponent(customOrderText)}`;
